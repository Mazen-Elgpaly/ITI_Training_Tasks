# Part 1 - Loops

### 1

- for...of
- forEach

# Part 9 - Mini Dashboard

اعمل Function اسمها

```js
dashboard()
```

لما تتنادى تطبع بالشكل ده

```
Total Employees : 6

Active Employees : ؟

Inactive Employees : ؟

IT Employees : ؟

Highest Salary : ؟

First HR Employee : ؟

Employee Names :

Ahmed
Sara
Ali
...
```

function dashboard() {
  const activeEmployees = employees.filter((employee) => employee.active);

  const inactiveEmployees = employees.filter((employee) => !employee.active);

  const itEmployees = employees.filter((employee) => {
    return employee.department === "IT";
  });

  const highestSalary = Math.max(
    ...employees.map((employee) => employee.salary)
  );

  const firstHrEmployee = employees.find((employee) => {
    return employee.department === "HR";
  });

  const employeeNames = employees.map((employee) => employee.name);

  console.log(`Total Employees : ${employees.length}`);

  console.log(`Active Employees : ${activeEmployees.length}`);

  console.log(`Inactive Employees : ${inactiveEmployees.length}`);

  console.log(`IT Employees : ${itEmployees.length}`);

  console.log(`Highest Salary : ${highestSalary}`);

  console.log(`First HR Employee : ${firstHrEmployee.name}`);

  console.log("Employee Names :");

  employeeNames.forEach((name) => {
    console.log(name);
  });
}

dashboard();

# Bonus

لو تعرف `reduce()`

بدون Loop عادية احسب

### 1

إجمالي المرتبات.
const totalSalary = employees.reduce((total, employee) => {
  return total + employee.salary;
}, 0);

console.log(totalSalary);
---

### 2

متوسط المرتبات.
const averageSalary = employees.reduce((total, employee) => {
  return total + employee.salary;
}, 0) / employees.length;

console.log(averageSalary);
---

### 3

أعلى مرتب.
const highestSalary = employees.reduce((highest, employee) => {
  return employee.salary > highest ? employee.salary : highest;
}, 0);

console.log(highestSalary);
---

### 4

عدد الموظفين النشطين.
const activeCount = employees.reduce((count, employee) => {
  return employee.active ? count + 1 : count;
}, 0);

console.log(activeCount);
